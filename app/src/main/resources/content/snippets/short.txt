# Written for lite-type.
static String season(int month) {
    return switch (month) {
        case 12, 1, 2 -> "winter";
        case 3, 4, 5 -> "spring";
        case 6, 7, 8 -> "summer";
        default -> "autumn";
    };
}

# Written for lite-type.
static int indexOf(int[] values, int target) {
    for (int i = 0; i < values.length; i++) {
        if (values[i] == target) {
            return i;
        }
    }
    return -1;
}

# Written for lite-type.
static String greeting(Optional<String> name) {
    return name.map(String::strip)
        .filter(n -> !n.isEmpty())
        .map(n -> "Hello, " + n + "!")
        .orElse("Hello!");
}
